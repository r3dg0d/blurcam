cmake_minimum_required(VERSION 3.24)
project(blurcam VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BLURCAM_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc imgcodecs objdetect video dnn)
find_package(CLI11 2.4 REQUIRED)
find_package(tomlplusplus 3.3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat>=60 libavcodec>=60 libavutil libswscale libavdevice)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
add_library(blurcam_core src/config.cpp src/effects.cpp src/detection.cpp src/tracking.cpp src/pipeline.cpp src/media.cpp src/live.cpp src/models.cpp)
target_include_directories(blurcam_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(blurcam_core PUBLIC ${OpenCV_LIBS} tomlplusplus::tomlplusplus PkgConfig::AV PkgConfig::SDL OpenSSL::Crypto CURL::libcurl Threads::Threads)
target_compile_options(blurcam_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(blurcam src/main.cpp)
target_link_libraries(blurcam PRIVATE blurcam_core CLI11::CLI11)
target_compile_options(blurcam PRIVATE -Wall -Wextra -Wpedantic)
if(BLURCAM_SANITIZE)
 target_compile_options(blurcam_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(blurcam_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(blurcam_tests tests/core.cpp)
 target_link_libraries(blurcam_tests PRIVATE blurcam_core)
 target_compile_definitions(blurcam_tests PRIVATE BLURCAM_FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
 add_test(NAME core COMMAND blurcam_tests)
 find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
 add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:blurcam>)
endif()
include(GNUInstallDirs)
install(TARGETS blurcam RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/blurcam.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES packaging/blurcam.bash DESTINATION ${CMAKE_INSTALL_DATADIR}/bash-completion/completions RENAME blurcam)
install(FILES packaging/_blurcam DESTINATION ${CMAKE_INSTALL_DATADIR}/zsh/site-functions)
install(FILES packaging/blurcam.fish DESTINATION ${CMAKE_INSTALL_DATADIR}/fish/vendor_completions.d)
set(CPACK_PACKAGE_VERSION "0.1.0")
set(CPACK_GENERATOR "TGZ")
include(CPack)
